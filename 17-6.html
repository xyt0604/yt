对应的代码 async 函数里的执行顺序与页面完全一致 
// 每一步：用 setTimeout 包装成一个 Promise
function xyt_step(xyt_name, xyt_ms) {
  return new Promise(function (xyt_resolve) {
    setTimeout(xyt_resolve, xyt_ms);
  });
}

async function xyt_makeDumplings() {
  await xyt_step("买原料", 6000);            // 6s 后 resolve
  await xyt_step("取原料", 3000);            // 接着 3s
  await Promise.all([                     // 同时开始，都完成才继续
    xyt_step("和面", 4000),
    xyt_step("和馅", 3000)
  ]);
  await xyt_step("包饺子", 5000);            // 两个都完成后 5s
  await xyt_step("吃饺子", 2000);
  console.log("全部完成！");
}
